import { createRoot } from 'react-dom/client';
import { useState } from 'react';
import { PrecisionCostTradeoff } from '../../../src/student/components/PrecisionCostTradeoff';
import { PrecisionPhase } from '../../../src/student/components/immersive-sample-planning/phases/PrecisionPhase';
import { FactorsPhase } from '../../../src/student/components/immersive-sample-planning/phases/FactorsPhase';
import { CVBasedSampleSizeCalculator } from '../../../src/student/components/CVBasedSampleSizeCalculator';
import '../../../src/student/student.css';

const base = { populationSize: 100, estimationType: 'mean' as const, confidenceLevel: '95' as const, marginOfError: 2, populationVariance: 100, designEffect: 1, nonResponseRate: 0, finalSampleSize: 50, recommendedSize: 50 };
function Review() {
  const [applied, setApplied] = useState('');
  const mode = new URLSearchParams(location.search).get('case');
  return <main className="mx-auto max-w-4xl p-6 bg-background text-foreground">
    <p className="mb-4 text-label">统计审查：隔离加载项目原组件，合成参数，无数据库连接</p>
    {mode === 'precision' ? <PrecisionPhase params={{ ...base, populationSize: 1000, populationVariance: 2e-10, marginOfError: 1e-6 }} onNext={() => {}} />
      : mode === 'fpc' ? <FactorsPhase step="cost_balance" params={base} onNext={() => {}} />
      : mode === 'cv' ? <CVBasedSampleSizeCalculator populationSize={100} confidenceLevel="95" estimationType="mean" initialDesignEffect={1} initialNonResponseRate={.5} onApplySampleSize={size => setApplied(`已应用 ${size}`)} />
      : <PrecisionCostTradeoff populationSize={100} confidenceLevel="95" designEffect={1} nonResponseRate={.5} initialMarginOfError={5} onApplyBudget={(_, size) => setApplied(`已应用 ${size}`)} />}
    <output id="review-applied" className="text-subheading">{applied}</output>
  </main>;
}
createRoot(document.getElementById('root')!).render(<Review />);
