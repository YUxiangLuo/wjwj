  onePercent: calculateSampleSizeMetrics({ populationSize: 100, confidenceLevel: '95', marginOfError: 1 })!.n,

// A cost tool can legitimately change confidence and target margin. The final
// analysis still reads the baseline fields, with no advanced-source resolution.
const advancedPopulation = Array.from({ length: 100 }, (_, i) => ({ event: i < 50 ? 'yes' : 'no', size: i + 1 }));
const appliedN = calculateSampleSizeMetrics({ populationSize: 100, confidenceLevel: '99', marginOfError: 1 })!.n;
const baselinePlan = { estimationType: 'proportion', confidenceLevel: '90', marginOfError: 10, designEffect: 1, nonResponseRate: 0,
  recommendedSize: calculateSampleSizeMetrics({ populationSize: 100, confidenceLevel: '90', marginOfError: 10 })!.n,
  finalSampleSize: appliedN, advancedPlanSource: 'cost', advancedPlans: [{ ...costSnapshot, inputs: { ...costSnapshot.inputs, confidenceLevel: '99', targetMarginOfError: 1, nonResponseRate: 0 } }] };
const advancedAnalysis = buildAnalysisMetrics({ population: advancedPopulation, fieldMapping: { numeric: ['size'], others: ['event'], analysisTarget: { kind: 'proportion', field: 'event', value: 'yes' } }, samplePlan: baselinePlan, samplingResults: { simple: { completed: true, sampleCount: appliedN, data: advancedPopulation.slice(0, appliedN) } } });
output.advancedAnalysisSource = { appliedConfidence: '99', appliedMargin: .01, targetPanelConfidence: advancedAnalysis.target?.confidenceLevel, targetPanelMargin: advancedAnalysis.target?.plannedMarginOfError,
  submitValidation: validateSamplePlanForSubmit(baselinePlan, 100, 'proportion') };

// Independent exact start distribution for systematic sampling: boundaries are
// integer multiples of 1/n. N equal-width intervals suffice, including N%n!=0.
let systematicChecks = 0; let maxSystematicDifference = 0;
for (let N = 2; N <= 40; N++) for (let n = 1; n <= N; n++) {
  const pop = Array.from({ length: N }, (_, i) => ({ y: ((i * 13 + N) % 17) - 8 }));
  const means = Array.from({ length: N }, (_, i) => {
    const start = (i + .5) / n;
    return Array.from({ length: n }, (_, j) => pop[Math.floor(start + j * N / n)]!.y).reduce((s, x) => s + x, 0) / n;
  });
  const mu = means.reduce((s, x) => s + x, 0) / N;
  const exactSe = Math.sqrt(means.reduce((s, x) => s + (x - mu) ** 2, 0) / N);
  const actual = computeSystematicExactMeanSe(pop, 'y', { completed: true, interval: N / n, sampleCount: n });
  if (actual === null) throw new Error(`Systematic unavailable N=${N}, n=${n}`);
  maxSystematicDifference = Math.max(maxSystematicDifference, Math.abs(actual - exactSe));
  systematicChecks++;
}
output.systematicIndependentChecks = { designs: systematicChecks, maxAbsoluteStandardErrorDifference: maxSystematicDifference };

// Translation sensitivity: design SE must be unchanged by adding a constant.
output.systematicTranslation = [0, 1e8, 1e10, 1e12].map(offset => {
  const pop = Array.from({ length: 100 }, (_, i) => ({ y: offset + (i % 7) * .01 }));
  const n = 30;
  const vals = Array.from({ length: 100 }, (_, i) => {
    const sample = performSystematicSampling(pop, n, () => (i + .5) / 100);
    return sample.data.reduce((sum, row) => sum + (row.y - offset), 0) / n;
  });
  const mean = vals.reduce((sum, value) => sum + value, 0) / vals.length;
  return { offset, currentSe: computeSystematicExactMeanSe(pop, 'y', { completed: true, interval: 100 / n, sampleCount: n }), centeredSe: Math.sqrt(vals.reduce((sum, value) => sum + (value - mean) ** 2, 0) / vals.length) };
});

const seeded = createPpsPopulation(); const seedPlan = computePpsMeanPlan(seeded);
const sigma2 = finitePopulationS2ToSigma2(seedPlan.populationVariance, seeded.length)!;
output.seedVariance = { storedVariance: seedPlan.populationVariance, trueSigma2: sigma2, currentN: seedPlan.recommendedSize, convertedN: calculateMeanEstimationSampleSize({ populationSize: seeded.length, confidenceLevel: '95', marginOfError: 2, populationVariance: sigma2 }) };

// Rare-influential PPS unit plus small background variation: report this as a
// residual Monte Carlo limitation, separate from the already-fixed zero check.
const rare = Array.from({ length: 1000 }, (_, i) => ({ y: i === 0 ? 1000 : (i % 2) * .01, size: i === 0 ? .001 : 1 }));
const rareContext = buildPpsStandardErrorContext(rare, 'size', 40, [{ key: 'y', accessor: row => Number(row.y) }], 512);
const pi = 40 * .001 / 999.001;
const rareSelected = performPPSSampling(rare, 'size', 40, () => pi / 2);
output.ppsResidualRisk = { mcSe: rareContext?.standardErrors.y, rarePi: pi, approximateChanceToMissRareIn512: (1 - pi) ** 512,
  rareSampleEstimate: estimateWeightedMean(rareSelected.data, 'y', rareSelected.inclusionProbabilities) };
const rareBoundaries = [0, 1]; let rareCumulative = 0;
for (const row of rare) {
  rareCumulative += 40 * row.size / 999.001;
  const fraction = rareCumulative - Math.floor(rareCumulative);
  if (fraction > 1e-10 && fraction < 1 - 1e-10) rareBoundaries.push(fraction);
}
rareBoundaries.sort((a, b) => a - b);
let rareMean = 0; let rareSecondMoment = 0; let rareMse = 0;
const rareTrueMean = rare.reduce((sum, row) => sum + row.y, 0) / rare.length;
for (let i = 1; i < rareBoundaries.length; i++) {
  const width = rareBoundaries[i]! - rareBoundaries[i - 1]!;
  if (width < 1e-10) continue;
  const sample = performPPSSampling(rare, 'size', 40, () => (rareBoundaries[i]! + rareBoundaries[i - 1]!) / 2);
  const estimate = estimateWeightedMean(sample.data, 'y', sample.inclusionProbabilities)!;
  rareMean += width * estimate; rareSecondMoment += width * estimate ** 2;
  rareMse += width * (estimate - rareTrueMean) ** 2;
}
Object.assign(output.ppsResidualRisk as object, { trueMean: rareTrueMean, exactDesignExpectation: rareMean,
  exactDesignSe: Math.sqrt(rareSecondMoment - rareMean ** 2), exactDesignRmse: Math.sqrt(rareMse) });

// Sample invariants for normal input ranges.
let ppsInvariants = 0;
for (let N = 2; N <= 40; N++) for (let n = 1; n <= N; n++) {
  const pop = Array.from({ length: N }, (_, i) => ({ y: i, size: 1 + ((i * 7) % 41) }));
  for (let seed = 0; seed < 5; seed++) {
    const sample = performPPSSampling(pop, 'size', n, createSeededRandom(seed));
    if (sample.data.length !== n || new Set(sample.indices).size !== n || sample.inclusionProbabilities.some(p => p <= 0 || p > 1)) throw new Error('PPS invariant failed');
    const simple = performSimpleSampling(pop, n, createSeededRandom(seed));
    if (simple.length !== n || new Set(simple).size !== n) throw new Error('SRS invariant failed');
    ppsInvariants++;
  }
}
output.sampleInvariants = { ppsAndSrsCases: ppsInvariants };

// Independent first-order pi construction followed by exact integration of all
// PPS systematic start intervals; checks the probabilities actually realized.
let ppsPiDesigns = 0; let maxPiDifference = 0;
for (let N = 2; N <= 30; N++) for (let n = 1; n <= N; n++) {
  const sizes = Array.from({ length: N }, (_, i) => 1 + ((i * 7) % 41));
  let left = 0; let right = n / Math.min(...sizes);
  for (let i = 0; i < 90; i++) {
    const lambda = (left + right) / 2;
    if (sizes.reduce((sum, size) => sum + Math.min(1, lambda * size), 0) < n) left = lambda; else right = lambda;
  }
  const pi = sizes.map(size => Math.min(1, right * size));
  const breaks = [0, 1]; let cumulative = 0;
  for (const p of pi) {
    if (p > 1 - 1e-9) continue;
    cumulative += p; const frac = cumulative - Math.floor(cumulative);
    if (frac > 1e-8 && frac < 1 - 1e-8) breaks.push(frac);
  }
  breaks.sort((a, b) => a - b);
  const observed = Array(N).fill(0);
  const pop = sizes.map((size, id) => ({ size, id }));
  for (let j = 1; j < breaks.length; j++) {
    const width = breaks[j]! - breaks[j - 1]!;
    if (width < 1e-8) continue;
    const draw = performPPSSampling(pop, 'size', n, () => (breaks[j]! + breaks[j - 1]!) / 2);
    for (const index of draw.indices) observed[index] += width;
  }
  maxPiDifference = Math.max(maxPiDifference, ...observed.map((p, i) => Math.abs(p - pi[i]!)));
  ppsPiDesigns++;
}
output.ppsExactPiChecks = { designs: ppsPiDesigns, maxAbsoluteInclusionProbabilityDifference: maxPiDifference };
const outputPath = new URL('./observations.json', import.meta.url);
await Bun.write(outputPath, JSON.stringify(output, null, 2) + '\n');
