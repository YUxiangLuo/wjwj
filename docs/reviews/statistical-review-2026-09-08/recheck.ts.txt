const out: Record<string, unknown> = {};
let assertions = 0;
function check(value: unknown, message: string): asserts value {
  assertions++;
  if (!value) throw new Error(message);
}
async function parse(csv: string) {
  return parseSpreadsheetFile(new TextEncoder().encode(csv).buffer, 'recheck.csv');
}
const methods: MethodKey[] = ['simple', 'systematic', 'pps', 'stratified'];
const mapping = { identifier: 'id', stratified: ['category'], numeric: ['size'], analysisTarget: { kind: 'proportion' as const, field: 'category', value: 'A' } };
const csv = 'id,category,size\n1,Ａ,1\n2,Ａ,2\n3,B,3\n4,B,4\n';
const parsed = await parse(csv);
check(validateTopicFieldMappingConfig(mapping) === null, 'F02 mapping config must be legal');
check(validateMappedPopulationData(parsed, mapping) === null, 'F02 uploaded population must be legal');
const sampled: SamplingResultsData = {};
for (const method of methods) {
  const action = await executeSamplingAction({ method, targetSampleSize: 4, population: parsed.rows, fieldMapping: mapping,
    rawParams: { sampleSize: 4, seed: 7, sizeField: 'size', stratifyBy: 'category', layers: [{ name: 'A', sampleCount: 2 }, { name: 'B', sampleCount: 2 }] } });
  check(action.ok, `F02 actual sampling action rejected ${method}`);
  sampled[method] = { ...action.result, serverGenerated: true } as any;
}
const replayed = hydrateSamplingResults(JSON.parse(JSON.stringify(compactSamplingResults(sampled))), parsed.rows);
check(replayed.ok, 'F02 stored results must replay');
const unicodeMetrics = buildAnalysisMetrics({ population: parsed.rows, fieldMapping: mapping, samplingResults: replayed.value });
for (const method of methods) check(Math.abs(unicodeMetrics.categorical.category!.representativeness[method]! - 100/3) < 1e-9, `F02 deviation changed for ${method}`);
out.F02 = { uploadedCategory: parsed.rows[0]?.category, uploadValidation: null, allSamplingActionsSucceeded: true, storageReplaySucceeded: true,
  population: unicodeMetrics.categorical.category!.population, sample: unicodeMetrics.categorical.category!.methods.simple, deviations: unicodeMetrics.categorical.category!.representativeness };
const asciiPopulation = parsed.rows.map(row=>({...row,category:String(row.category).normalize('NFKC')}));
const asciiReplayed = hydrateSamplingResults(compactSamplingResults(sampled),asciiPopulation);
check(asciiReplayed.ok,'F02 ASCII control replays');
const asciiMetrics = buildAnalysisMetrics({population:asciiPopulation,fieldMapping:mapping,samplingResults:asciiReplayed.value});
for (const method of methods) check(asciiMetrics.categorical.category!.representativeness[method]===0,'F02 changing only category normalization should remove error');
out.F02Control = { asciiCensusDeviations:asciiMetrics.categorical.category!.representativeness };

// A legitimate systematic random draw makes the tail counterexample common:
// first 19 category blocks have 20 rows; tail alternates c20/c21.
const tailRows = Array.from({length:420}, (_,i) => ({ id: i + 1, category: i < 380 ? `c${Math.floor(i / 20) + 1}` : (i % 2 === 0 ? 'c20' : 'c21') }));
const tail = await parse('id,category\n' + tailRows.map(row => `${row.id},${row.category}`).join('\n') + '\n');
const tailMapping = { identifier: 'id', others: ['category'], analysisTarget: {kind:'proportion' as const, field:'category', value:'c21'} };
check(validateTopicFieldMappingConfig(tailMapping) === null, 'F03 mapping config must be legal');
check(validateMappedPopulationData(tail, tailMapping) === null, 'F03 population must be legal');
let selected: any;
for (let seed = 0; seed < 20; seed++) {
  const action = await executeSamplingAction({method:'systematic', targetSampleSize:210, population:tail.rows, fieldMapping:tailMapping, rawParams:{seed}});
  check(action.ok, 'F03 systematic action must succeed');
  if (action.result.data.every(row => row.category !== 'c21')) { selected = action.result; break; }
}
check(selected, 'F03 obtain an actual seed that misses c21');
const tailMetrics = buildAnalysisMetrics({ population:tail.rows, fieldMapping:tailMapping, samplingResults:{systematic:selected} });
const originalCategories = Array.from({length:21}, (_,i) => `c${i+1}`);
const fullDifference = originalCategories.reduce((sum,c) => sum + Math.abs(selected.data.filter((r:any)=>r.category===c).length/210 - 1/21)*100,0)/21;
check(tailMetrics.categorical.category!.representativeness.systematic === 0, 'F03 reported discrepancy must be zero');
check(fullDifference > 0.45, 'F03 original discrepancy must be positive');
out.F03 = { uploadValidation:null, actualSamplingMethod:'systematic', seed:selected.seed, randomStart:selected.randomStart, sampleCount:selected.sampleCount,
  c20Count:selected.data.filter((r:any)=>r.category==='c20').length, c21Count:selected.data.filter((r:any)=>r.category==='c21').length,
  fullCategoryDifferencePercentagePoints:fullDifference, reportedDifference:tailMetrics.categorical.category!.representativeness.systematic,
  truncated:tailMetrics.categorical.category!.truncated, targetPopulation:tailMetrics.target!.populationValue, targetEstimate:tailMetrics.target!.methods.systematic.estimate };

const record = { id:1, version:1, status:'in_progress', currentStage:2, stage1CompletedAt:new Date('2026-09-01T00:00:00Z') } as any;
const policy = {version:1,methods:{simple:'required',systematic:'required',pps:'required',stratified:'required'}};
const stage = (samplePlan:any) => buildStageCompletionUpdates({ experimentRecord:record, stage:2, payload:{samplePlan}, populationSize:100,
  deadlinePassed:false, fieldMapping:mapping, samplingPolicy:policy, stratifiedStats:{category:{A:50,B:50}} });
const metrics = calculateSampleSizeMetrics({populationSize:100,confidenceLevel:'95',marginOfError:5,designEffect:1,nonResponseRate:.5})!;
const cost = {version:1 as const,kind:'cost' as const,appliedAt:'2026-09-08T00:00:00Z', inputs:{confidenceLevel:'95',targetMarginOfError:5,designEffect:1,nonResponseRate:.5,fixedCost:5000,variableCostPerUnit:15,totalBudget:50000},
  result:{requiredSampleSize:100,totalCost:6500,feasible:true}};
const base = {estimationType:'proportion',confidenceLevel:'95',marginOfError:5,designEffect:1,nonResponseRate:.5,recommendedSize:100,finalSampleSize:100,advancedPlanSource:'cost',advancedPlans:[cost]};
const capSaved = stage(base);
check(capSaved.ok, 'F01 actual stage completion must accept capped snapshot');
const independentMe = 1.96*Math.sqrt(.25/50*(100-50)/(100-1));
check(Math.abs(metrics.marginOfError-independentMe)<1e-12, 'F01 formula must independently agree');
check(independentMe>.05, 'F01 target really unmet');
out.F01 = { stageCompletionAccepted:capSaved.ok, inviteSize:100, expectedCompleted:50, independentMarginOfError:independentMe, targetMarginOfError:.05, budget:50000,totalCost:6500 };
const noNonresponse = calculateSampleSizeMetrics({populationSize:100,confidenceLevel:'95',marginOfError:5,designEffect:1,nonResponseRate:0})!;
check(noNonresponse.marginOfError<=.05,'F01 no-nonresponse control reaches target');
out.F01Control = {nonResponseRate:0,inviteSize:noNonresponse.n,marginOfError:noNonresponse.marginOfError};
const invalid = {...cost,inputs:{...cost.inputs,targetMarginOfError:-.05,designEffect:-2,nonResponseRate:2,fixedCost:-1000,variableCostPerUnit:-5,totalBudget:-500},result:{requiredSampleSize:80,totalCost:-1400,feasible:true}};
check(parseAdvancedPlans([invalid]) !== null, 'F04 malformed numeric assumptions parse');

const sourcePlan = {...base,confidenceLevel:'90',marginOfError:10,nonResponseRate:0,recommendedSize:calculateSampleSizeMetrics({populationSize:100,confidenceLevel:'90',marginOfError:10})!.n,advancedPlans:[sourceCost]};

const tinyText = 'id,y\n' + Array.from({length:1000},(_,i) => `${i+1},${i%2===0 ? 0 : Math.sqrt(8e-10)}`).join('\n') + '\n';

// R01 independent PPS exact distribution: integer cumulative probability
// numerators and explicit threshold selection, without calling the PPS sampler
// or weighted-estimator implementation for the reference value.
const D = 999001;
const probabilityNumerators = [40,...Array(999).fill(40000)] as number[];
const cutNumerators = new Set<number>([0,D]);
let cumulativeNumerator = 0;
for (const p of probabilityNumerators) { cumulativeNumerator += p; cutNumerators.add(cumulativeNumerator % D); }
const cuts = [...cutNumerators].sort((a,b)=>a-b);
let expectation = 0, second = 0, weightSum = 0;
for(let k=1;k<cuts.length;k++) {
  const weight = (cuts[k]!-cuts[k-1]!)/D;
  const start = (cuts[k]!+cuts[k-1]!)/(2*D);
  let cumulative = probabilityNumerators[0]!, index=0, numerator=0, denominator=0;
  for(let j=0;j<40;j++) {
    const threshold = (start+j)*D;
    while(cumulative<=threshold) cumulative+=probabilityNumerators[++index]!;
    const y = index===0 ? 1000 : (index%2)*.01;
    numerator += y/probabilityNumerators[index]!;
    denominator += 1/probabilityNumerators[index]!;
  }
  const estimate = numerator/denominator;
  expectation += weight*estimate; second += weight*estimate*estimate; weightSum += weight;
}
const independentSe = Math.sqrt(second-expectation**2);
check(Math.abs(weightSum-1)<1e-12,'R01 all exact start intervals cover probability one');
check(Math.abs(independentSe-6.090046666003953)<1e-8,'R01 independent sampler reproduces exact SE');
const rarePopulation = Array.from({length:1000},(_,i)=>({id:i+1,y:i===0?1000:(i%2)*.01,size:i===0?.001:1}));
const rareMapping = {identifier:'id',numeric:['y','size'],analysisTarget:{kind:'mean' as const,field:'y'}};
const rareParsed = await parse('id,y,size\n'+rarePopulation.map(r=>`${r.id},${r.y},${r.size}`).join('\n')+'\n');
check(validateMappedPopulationData(rareParsed,rareMapping)===null,'R01 rare-unit population is upload-valid');
const rareAction = await executeSamplingAction({method:'pps',targetSampleSize:40,population:rareParsed.rows,fieldMapping:rareMapping,rawParams:{seed:0,sampleSize:40,sizeField:'size'}});
check(rareAction.ok,'R01 PPS actual action succeeds');
const rareMetrics = buildAnalysisMetrics({population:rareParsed.rows,fieldMapping:rareMapping,samplingResults:{pps:rareAction.result},samplePlan:{confidenceLevel:'95',marginOfError:1}});
check(rareMetrics.target!.methods.pps.reliability.status==='caution','R01 existing caution must be acknowledged');
out.R01 = { integerDenominator:D,intervalCount:cuts.length-1,probabilityWeightSum:weightSum,independentExpectation:expectation,independentSe,
  actualAnalysis:rareMetrics.target!.methods.pps };
